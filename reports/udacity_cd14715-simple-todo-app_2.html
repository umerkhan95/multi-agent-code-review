After:

export function sanitizeInput(input: string): string {
  const htmlEntities: Record<string, string> = {
    '<': '&lt;',
    '>': '&gt;',
    '"': '&quot;',
    "'": '&#039;'
  } as const;
  
  return input.replace(/[<>"']/g, (char) => htmlEntities[char] ?? char);
}
